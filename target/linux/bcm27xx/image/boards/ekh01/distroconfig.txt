# Restore PL011 (ttyAMA0) to GPIOs 14 & 15, instead of Mini UART (ttyS0).
# Mini UART is disabled by default unless "enable_uart=1" is specified,
#  which changes the core frequency to a fixed value and impacts performance.
# See https://www.raspberrypi.com/documentation/computers/configuration.html#mini-uart-and-cpu-core-frequency
[pi0w]
dtoverlay=disable-bt
[pi3]
dtoverlay=disable-bt
dtparam=i2c1=on
# Pin the core clock (Batman #203 risk 2): on BCM283x the SPI clock is derived
# from core_freq, and on a Pi 3 core_freq scales down with load - measured at
# 287.5 MHz while idle on a 3A+, against core_freq=400. A drifting core clock
# means a drifting MM6108 SPI clock. Verified pinned at 400 MHz after this.
core_freq_min=400
[pi4]
dtoverlay=miniuart-bt
dtparam=i2c1=on
[cm4]
# This overlay is used to allow m.2 wifi cards
dtoverlay=pcie-32bit-dma
# Enable USB ports on carrier cards
dtoverlay=dwc2,dr_mode=host

[all]
# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1

# How much memory in MB to assign to the GPU on Pi models having
# 256, 512 or 1024 MB total memory
gpu_mem_256=128
gpu_mem_512=128
gpu_mem_1024=128

# enable uart
uart_enable=1

# enable uart5
dtoverlay=uart5

# disable SD activity LED by default
#(we configure it for HaLow later in userspace)
dtparam=act_led_trigger=none

# adds ramoops device tree overlay
# Board-conditional (#173 / Batman #203): the reserved-memory reg property must
# match the SoC's #address-cells. ramoops-pi4 carries a 3-cell reg (64-bit
# address) which is correct on bcm2711; on bcm2710 the DT expects 2 cells and the
# node is rejected at boot --
#   OF: fdt: Reserved memory: invalid reg property in 'ramoops@b000000', skipping node
# -- leaving the board with no crash black-box at all. The generic ramoops overlay
# has the 2-cell reg; bench-verified on a Pi 3A+ 2026-09-29 (sysrq crash captured to
# pstore, dmesg-ramoops-0 contained the pre-panic marker).
# NB bcm2710 devices reuse this file via IMAGE/sysupgrade := ... | boot-2710 | boot-ekh01.
[pi4]
dtoverlay=ramoops-pi4,console-size=0x8000
[pi3]
dtoverlay=ramoops,console-size=0x8000
[pi02]
dtoverlay=ramoops,console-size=0x8000
[all]

# automatically load sensor dtoverlays
camera_auto_detect=1
